# AI Video Assistant - Python dependencies
#
# Install with:   pip install -r requirements.txt
#
# FFmpeg is also required but is NOT a pip package. It is a system program.
#   Windows: winget install Gyan.FFmpeg
#   macOS:   brew install ffmpeg
#   Linux:   sudo apt install ffmpeg


# ─── Web interface ───────────────────────────────────────────────────────────
streamlit                     # app.py - the whole UI
markdown                      # app.py - renders LLM markdown inside HTML cards


# ─── Audio: download, convert, chunk ─────────────────────────────────────────
yt-dlp                        # utils/audio_processor.py - YouTube audio download
pydub                         # utils/audio_processor.py, core/transcriber.py - WAV conversion and splitting


# ─── Speech to text ──────────────────────────────────────────────────────────
openai-whisper                # core/transcriber.py - local English transcription with segment timings
requests                      # core/transcriber.py - calls the Sarvam AI Hinglish endpoint


# ─── LLM orchestration ───────────────────────────────────────────────────────
langchain-core                # prompts, output parsers, Runnable chains, Document
langchain-text-splitters      # RecursiveCharacterTextSplitter
langchain-mistralai           # ChatMistralAI - summary, extraction, grounded chat


# ─── Embeddings and vector store ─────────────────────────────────────────────
langchain-chroma              # core/vector_store.py - Chroma wrapper
langchain-huggingface         # core/vector_store.py - HuggingFaceEmbeddings
sentence-transformers         # loads the all-MiniLM-L6-v2 embedding model


# ─── Configuration ───────────────────────────────────────────────────────────
python-dotenv                 # core/config.py - reads the .env file


# ─── Compatibility ───────────────────────────────────────────────────────────
# Python 3.13 removed the "audioop" module from the standard library, and pydub
# imports it. This backport restores it. Ignored on Python 3.12 and older.
audioop-lts; python_version >= "3.13"


# ─── Installed automatically as dependencies of the above ────────────────────
# torch and numba        (via openai-whisper)      - large download, ~2 GB
# chromadb               (via langchain-chroma)
# transformers, huggingface-hub, numpy, scipy  (via sentence-transformers)
# langchain-core         (via the langchain-* packages)
#
# There is no need to list them here. pip resolves them, and pinning them by
# hand is the usual cause of version conflicts.
